Примечание:
1- pydivert должен быть установлен: pip install pydivert
2- python должен быть установлен 3.8 или выше
3- Остановка обхода — CTRL+C
4- Обход не идеален, могут быть проблемы с отдельными сайтами
5- Для новичков лучше EXE-версия

# ──────────── Базовые режимы ────────────

# Всё по умолчанию (disorder, midsld, встроенные домены)
python pydpi.py start

# Самый простой режим
python pydpi.py start --mode split

# Disorder + разрыв в середине домена второго уровня (по умолчанию)
python pydpi.py start --mode disorder --split midsld

# Split + случайная точка разрыва
python pydpi.py start --mode split --split random

# Fake + split (рекомендуется, если disorder не пробивает)
python pydpi.py start --mode fake+split

# Fake + disorder (самый «злой»)
python pydpi.py start --mode fake+disorder

# ──────────── Параметры фейка ────────────

# TTL фейка = 3 (по умолчанию 1)
python pydpi.py start --mode fake+disorder --ttl 3

# Отправить фейк 4 раза (по умолчанию 2)
python pydpi.py start --mode fake+split --repeats 4

# ──────────── Домены ────────────

# Обходить ВСЁ,(осторожно)
python pydpi.py start --all --mode disorder

# Обход всего с fake+split
python pydpi.py start --all --mode fake+split

# ──────────── Диагностика ────────────

# Проверить, что WinDivert открывается (5 сек)
python pydpi.py test


# Проверить DNS на подмену (нужен интернет)
python pydpi.py dns

# ──────────── Справка ────────────

# Посмотреть все режимы и примеры
python pydpi.py list-modes

# Версия
python pydpi.py --version

# Полная справка
python pydpi.py --help
python pydpi.py start --help
python pydpi.py test --help
python pydpi.py dns --help
python pydpi.py list-modes --help

# Рекомендуемая последовательность (если не пробивает) 

# 1. Дефолт
python pydpi.py start

# 2. Если Discord/YouTube не открываются — fake+split
python pydpi.py start --mode fake+split

# 3. Если fake не проходит — disorder без фейка
python pydpi.py start --mode disorder

# 4. Если и disorder не пробивает — fake+disorder
python pydpi.py start --mode fake+disorder --ttl 2 --repeats 3

# 5. Если совсем ничего — обход всего
python pydpi.py start --all --mode fake+disorder
